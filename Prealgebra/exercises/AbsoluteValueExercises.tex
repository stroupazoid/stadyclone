\section*{Reading Expressions from the Inside Out}

\begin{problem}
A computer evaluates the expression below from the inside outward.

\[
-(-5)
\]

First identify the number being acted on: $-5$.

The outside negative sign means ``take the opposite.''

\[
-(-5)=\answer{5}
\]

\end{problem}


\begin{problem}
Evaluate the expression from the inside outward.

\[
-|-4|
\]

First evaluate the absolute value:

\[
|-4|=\answer{4}
\]

Now take the opposite:

\[
-(4)=\answer{-4}
\]

\end{problem}


\begin{problem}
Evaluate the expression from the inside outward.

\[
-|-(-3)|
\]

First evaluate the innermost operation:

\[
-(-3)=\answer{3}
\]

Next take the absolute value:

\[
|3|=\answer{3}
\]

Finally take the opposite:

\[
-(3)=\answer{-3}
\]

\end{problem}


\section*{Error Analysis}

\begin{problem}
A student writes

\[
-|-8|=8.
\]

What mistake did the student make?

\begin{multipleChoice}
\choice{The student evaluated the absolute value incorrectly.}
\choice[correct]{The student forgot that the negative sign outside the absolute value means to take the opposite after finding the absolute value.}
\choice{The student should have changed $-8$ to $-16$.}
\end{multipleChoice}

Now evaluate the expression correctly.

\[
-|-8|=\answer{-8}
\]

\end{problem}


\section*{Application}

\begin{problem}
A machine is supposed to cut a metal rod to exactly $25$ centimeters. A rod is cut to $24.6$ centimeters.

The error is

\[
24.6-25=\answer{-0.4}.
\]

The absolute error is

\[
|24.6-25|=\answer{0.4}.
\]

Which statement best describes the result?

\begin{multipleChoice}
\choice{The rod is $0.4$ centimeters too long.}
\choice[correct]{The rod is $0.4$ centimeters too short.}
\choice{The rod is exactly the correct length.}
\end{multipleChoice}

\end{problem}
